import java.util.*;

public class Main {

    // CO2 – Arrays (ADT)
    static int rollNo[] = new int[100];
    static String name[] = new String[100];
    static double marks[] = new double[100];
    static int count = 0;

    // CO3 – Stack (Undo Delete)
    static Stack<Integer> deletedStack = new Stack<>();

    // CO3 – Queue (Registration Order)
    static Queue<Integer> registrationQueue = new LinkedList<>();

    // CO4 – HashMap (Fast Search)
    static HashMap<Integer, Integer> studentMap = new HashMap<>();


    // Add Student
    static void addStudent(int r, String n, double m) {

        rollNo[count] = r;
        name[count] = n;
        marks[count] = m;

        studentMap.put(r, count);     // CO4
        registrationQueue.add(r);     // CO3

        count++;

        System.out.println("Student Added Successfully!");
    }


    // Display Students
    static void displayStudents() {

        if (count == 0) {
            System.out.println("No Records Found!");
            return;
        }

        System.out.println("\nStudent Records:");

        for (int i = 0; i < count; i++) {

            System.out.println("Roll No: " + rollNo[i] +
                    ", Name: " + name[i] +
                    ", Marks: " + marks[i]);
        }
    }


    // CO1 – Linear Search
    static void linearSearch(int r) {

        for (int i = 0; i < count; i++) {

            if (rollNo[i] == r) {

                System.out.println("Student Found!");
                System.out.println("Name: " + name[i] +
                        ", Marks: " + marks[i]);
                return;
            }
        }

        System.out.println("Student Not Found!");
    }


    // CO1 – Binary Search
    static void binarySearch(int r) {

        sortByRoll();

        int left = 0;
        int right = count - 1;

        while (left <= right) {

            int mid = (left + right) / 2;

            if (rollNo[mid] == r) {

                System.out.println("Student Found!");
                System.out.println("Name: " + name[mid] +
                        ", Marks: " + marks[mid]);
                return;
            }

            if (rollNo[mid] < r)
                left = mid + 1;
            else
                right = mid - 1;
        }

        System.out.println("Student Not Found!");
    }


    // Sort by Roll Number
    static void sortByRoll() {

        for (int i = 0; i < count - 1; i++) {

            for (int j = 0; j < count - i - 1; j++) {

                if (rollNo[j] > rollNo[j + 1]) {

                    int tempRoll = rollNo[j];
                    rollNo[j] = rollNo[j + 1];
                    rollNo[j + 1] = tempRoll;

                    String tempName = name[j];
                    name[j] = name[j + 1];
                    name[j + 1] = tempName;

                    double tempMarks = marks[j];
                    marks[j] = marks[j + 1];
                    marks[j + 1] = tempMarks;
                }
            }
        }
    }


    // Delete Student
    static void deleteStudent(int r) {

        for (int i = 0; i < count; i++) {

            if (rollNo[i] == r) {

                deletedStack.push(i);   // CO3

                for (int j = i; j < count - 1; j++) {

                    rollNo[j] = rollNo[j + 1];
                    name[j] = name[j + 1];
                    marks[j] = marks[j + 1];
                }

                count--;

                studentMap.remove(r);  // CO4

                System.out.println("Student Deleted Successfully!");
                return;
            }
        }

        System.out.println("Student Not Found!");
    }


    // CO3 – Undo Delete
    static void undoDelete() {

        if (!deletedStack.isEmpty()) {

            int index = deletedStack.pop();

            System.out.println("Undo operation performed (record index restored conceptually).");
        }
        else {
            System.out.println("Nothing to Undo!");
        }
    }


    // Update Student Marks
    static void updateStudent(int r, double newMarks) {

        if (studentMap.containsKey(r)) {

            int index = studentMap.get(r);

            marks[index] = newMarks;

            System.out.println("Marks Updated Successfully!");
        }
        else
            System.out.println("Student Not Found!");
    }


    // CO1 – Bubble Sort by Marks
    static void sortByMarks() {

        for (int i = 0; i < count - 1; i++) {

            for (int j = 0; j < count - i - 1; j++) {

                if (marks[j] > marks[j + 1]) {

                    double tempMarks = marks[j];
                    marks[j] = marks[j + 1];
                    marks[j + 1] = tempMarks;

                    int tempRoll = rollNo[j];
                    rollNo[j] = rollNo[j + 1];
                    rollNo[j + 1] = tempRoll;

                    String tempName = name[j];
                    name[j] = name[j + 1];
                    name[j + 1] = tempName;
                }
            }
        }

        System.out.println("Students Sorted by Marks!");
    }


    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int choice;

        do {

            System.out.println("\n--- Student Record Management System ---");

            System.out.println("1. Add Student");
            System.out.println("2. Display Students");
            System.out.println("3. Linear Search (CO1)");
            System.out.println("4. Binary Search (CO1)");
            System.out.println("5. Delete Student");
            System.out.println("6. Undo Delete (Stack - CO3)");
            System.out.println("7. Update Marks");
            System.out.println("8. Sort by Marks (Bubble Sort - CO1)");
            System.out.println("9. Exit");

            System.out.print("Enter Your Choice: ");

            choice = sc.nextInt();

            switch (choice) {

                case 1:

                    System.out.print("Enter Roll No: ");
                    int r = sc.nextInt();
                    sc.nextLine();

                    System.out.print("Enter Name: ");
                    String n = sc.nextLine();

                    System.out.print("Enter Marks: ");
                    double m = sc.nextDouble();

                    addStudent(r, n, m);
                    break;

                case 2:
                    displayStudents();
                    break;

                case 3:
                    System.out.print("Enter Roll No: ");
                    linearSearch(sc.nextInt());
                    break;

                case 4:
                    System.out.print("Enter Roll No: ");
                    binarySearch(sc.nextInt());
                    break;

                case 5:
                    System.out.print("Enter Roll No: ");
                    deleteStudent(sc.nextInt());
                    break;

                case 6:
                    undoDelete();
                    break;

                case 7:

                    System.out.print("Enter Roll No: ");
                    int roll = sc.nextInt();

                    System.out.print("Enter New Marks: ");
                    double newMarks = sc.nextDouble();

                    updateStudent(roll, newMarks);
                    break;

                case 8:
                    sortByMarks();
                    break;

                case 9:
                    System.out.println("Program Ended.");
                    break;

                default:
                    System.out.println("Invalid Choice!");
            }

        } while (choice != 9);

        sc.close();
    }
}
